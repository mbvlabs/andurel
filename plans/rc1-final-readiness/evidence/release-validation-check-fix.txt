+func checkNarsilcRuntime(rootDir string) checkResult {
+	const name = "narsilc runtime"
+
+	lock, err := layout.ReadLockFile(rootDir)
+	if err != nil {
+		return checkResult{
+			name:    name,
+			status:  statusSkip,
+			message: "andurel.lock missing or invalid (skipping narsilc runtime check)",
+		}
+	}
+	tool, configured := lock.Tools["narsilc"]
+	if !configured || tool == nil {
+		return checkResult{
+			name:    name,
+			status:  statusPass,
+			message: "narsilc is not configured in andurel.lock",
+		}
+	}
+
+	state := inspectNarsilcRuntime(rootDir, tool.Version)
+	switch state.Status {
+	case narsilcRuntimeAligned:
+		return checkResult{
+			name:    name,
+			status:  statusPass,
+			message: fmt.Sprintf("go.mod matches narsilc tool %s", state.ToolVersion),
+		}
+	case narsilcRuntimeMissing:
+		return checkResult{
+			name:   name,
+			status: statusPass,
+			message: fmt.Sprintf(
+				"narsilc tool %s is configured; go.mod has no narsilc runtime requirement",
+				state.ToolVersion,
+			),
+		}
+	case narsilcRuntimeMismatch:
+		details := []string{state.Detail, "Run:"}
+		details = append(details, state.remediation()...)
+		return checkResult{
+			name:   name,
+			status: statusWarn,
+			message: fmt.Sprintf(
+				"narsilc tool %s, go.mod requires %s",
+				state.ToolVersion,
+				state.RequiredVersion,
+			),
+			details: details,
+			hint: fmt.Sprintf(
+				"Run '%s', then 'go mod tidy' to align the runtime with the narsilc tool.",
+				state.remediation()[0],

+// isGoCacheDataFile reports whether path has the go command's build-cache
+// data-artifact layout: <cache>/<first two hex digits>/<64-hex action ID>-d.
+// Ordinary project source, including .go files with generated headers, does not

+func isGoCacheDataFile(cacheDir, path string) bool {
+	rel, ok := relativeWithinDir(cacheDir, path)
+	if !ok {
+		return false
+	}
+	parts := strings.Split(filepath.ToSlash(rel), "/")
+	if len(parts) != 2 {
+		return false
+	}
+	prefix, name := parts[0], parts[1]
+	if len(prefix) != 2 || !isLowerHex(prefix) {
+		return false

+		return false
+	}

+		if (c < '0' || c > '9') && (c < 'a' || c > 'f') {
+			return false
+		}
+	}

+func relativeWithinDir(dir, path string) (string, bool) {
